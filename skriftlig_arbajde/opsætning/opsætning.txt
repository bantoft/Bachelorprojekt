qrsh

##############
#Load byg dev#
##############
cd simulatorer/BOUT

cd BOUT-dev
git checkout v4.3.2
./configure
make -j16


################
#Load byg hesel#
################
cd simulatorer/BOUT

git clone --recursive git@github.com:PPFE-Turbulence/BOUT-HESEL.git
cd BOUT-HESEL

git config --file=.gitmodules submodule.HESEL-Common.url git@github.com:PPFE-Turbulence/HESEL-Common.git
git submodule sync
git submodule update --init --recursive

ln -s HESEL-Common/BoutFastOutput BoutFastOutput
ln -s HESEL-Common/HeselParameters HeselParameters
ln -s HESEL-Common/Neutrals Neutrals
ln -s HESEL-Common/Parallel Parallel
make -j16

################
#Kør simulation#
################
cd simulator/BOUT-HESEL
mpirun -np 4 ./hesel -d data
