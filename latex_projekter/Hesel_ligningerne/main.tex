\section{Standard form}

HESEL ligningerne er givet på følgende form:

\begin{gather*}
    \mathrm{d} _t n + n\mathcal{K}(\phi) - \mathcal{K}(p_e) = \Lambda_n  \label{tæthedsligning} \\ 
    \mathrm{d}_t^0 \omega^* + \left\{\phi, p_i \right\} - \mathcal{K}(p_e+p_i)=\Lambda_{\omega^*} \label{vorticiteten} \\
    \frac{3}{2}\mathrm{d} _t p_e + \frac{5}{2}p_e \mathcal{K}(\phi) -\frac{5}{2}\mathcal{K}\left(\frac{p_e^2}{n}\right)= \Lambda_{p_e}\label{elektrontryksligning} \\
    \frac{3}{2}\mathrm{d} _t p_i + \frac{5}{2}p_i \mathcal{K}(\phi) +\frac{5}{2}\mathcal{K}\left(\frac{p_i^2}{n}\right) - p_i\mathcal{K}(p_e+p_i) = \Lambda_{p_i} \label{iontryksligning}
\end{gather*}


\section{Lambda termer}
Lambda termerne er givet ved:
\begin{align*}
\Lambda_n =& D_e\nabla_\perp^2 n -\frac{n}{\tau} -\frac{n-n_p}{\tau_p}-\alpha\left(\tilde T_e+ \frac{\tilde T_e}{\bar n}\tilde n-\tilde\phi\right)\\ 
\Lambda_\omega^{*}=& D_i\nabla_\perp^2\omega^{*}-\frac{\omega^{*}}{\tau}+\frac{\rho_s}{L_c}\left[1-\exp\left(\phi_m-\frac{\phi_s}{T_{e,s}}\right)\right] \\
                   &-\alpha\left( \tilde T_e+ \frac{\tilde T_e}{\bar n}\tilde n -\tilde\phi \right)\\
\Lambda_{p_e} =& \frac{5}{2}D_e\nabla_\perp^2 p_e  + \left(\frac{16}{6} - \frac{5}{2}\right)\nabla\cdot\left(n\nabla_\perp T_e\right) - \frac{9p_e}{2\tau} - \frac{T_e}{\tau^{SH}}-\Theta -\frac{p_e-p_{e,p}}{\tau_p} \\
               &-\bar T_e \alpha\left( \tilde T_e + \frac{\tilde T_e}{\bar n}\tilde n -\tilde\phi \right)\\
\Lambda_{p_i} =& D_i\nabla_\perp^2 p_i - D_iT_i\nabla_\perp^2n - \frac{9p_i}{2\tau} +\Theta -\frac{p_i-p_{i,p}}{\tau_p} +p_i\Lambda_{\omega^*}
\end{align*}


\section{Koeffecienter}
Samling af koeffecienter:
\begin{gather*}
    T_{e,i} = p_{e,i}/n \\
    D_i = \left(1 + \frac{R}{a}q^2\right)\frac{\rho_{i0}^2v_{ii,0}}{\rho_{s0}^2\omega_{i0}}\\
    D_e = \left(1 + \frac{T_{i0}}{T_{e0}}\right) \left(1 + \frac{R}{a}q^2\right)\frac{\rho_{e0}^2v_{ei,0}}{\rho_{s0}^2\omega_{i0}}\\
    \Theta = \frac{3m_e}{m_i}v_{ei}\left(p_e - p_i\right)\\
    \tau = \frac{L_b}{2M_\parallel cs} \\
    \tau^{SH} = \frac{L_b}{X_{\parallel,e}^{SH}} \\
    \alpha = \frac{2T_{e0}}{v_{ei}m_eL^2_\parallel\omega_{i0}}\\
    c_s = \sqrt{(T_e + T_i)/m_i} \\
    \phi_m = \ln(\sqrt{m_i/2\pi m_e})\\
    \L_\parallel = L_b = qR\\
    \rho_s = \sqrt{\frac{T_{e0}}{m_i\Omega_{ei}^2}}\\
\end{gather*}


\section{Forkotelser}
De forkortede symboler er givet ved:
\begin{align*}
\omega^* = \nabla^2\phi + \nabla^2 p_i
\end{align*}

\section{Operatorer}
Herved er operatorerne defineret med:
\begin{gather*}
    \mathrm{d}_t^0 f = \partial_t f + \left\{\phi, f \right\}\\
    \mathrm{d}_t f = \partial_t f  + \frac{1}{B}  \left\{ \phi , f \right\} \\    
    \left\{ f, g\right\} = \partial_x f \partial_y g - \partial_y f \partial_x g\\
    \mathcal{K}(f)= - \frac{\rho_s}{R}\partial_y f \\
    \bar f = \frac{1}{L_y}\int_0^{L_y} f \mathrm{d}y\\
    \tilde f = f - \bar f\\
    \nabla = \left(\partial_x, \partial_y, \partial_z \right)^T\\
    \nabla_\perp = \left(\partial_x, \partial_y, 0 \right)^T\\
    \nabla^2 = \left(\partial^2_x, \partial^2_y, \partial^2_z \right)^T\\
    \nabla_\perp^2 = \left(\partial^2_x, \partial^2_y, 0 \right)^T\\
\end{gather*}


\section{Funktioner}
Funktionerne er gevet ved:
\begin{align*}
    &n: n(x,y,t): \quad "Tæthed" \\
    &p_e: p_e(x,y,t): \quad "Elektrontryk" \\
    &p_i: p_i(x,y,t): \quad "Iontryk" \\
    &\phi: \phi(x,y,t): \quad "Elektrisk potentiale" \\
\end{align*}

\section{parametre}
\begin{align*}
    L_b = qR\approx8m\\
    L_c \approx 15m\\
    T_{e0} = 25eV\\
    T_{i0} = 25eV\\
    M_\parallel = 0.5\\
    2M_\parallel c_s  = 50000ms^{-1}\\
    c_s = \sqrt{(T_e + T_i)/m_i}\\
\end{align*}
